getInterp <- function(values, plotres=0.5, xlim, ylim){
  require(akima)
  #plotres <- 0.5 #  0.02 # 0.05 # 0.2 resolution of interpolation grid, km
  xlength <- (xlim[2]-xlim[1])/plotres
  ylength <- (ylim[2]-ylim[1])/plotres
  xx <- nodes # 1:nrow(coordsn) #which(coordsn$X <= xlim[2] & coordsn$X >= xlim[1] & coordsn$Y <= ylim[2] & coordsn$Y >= ylim[1])
  x=coordsn$X[xx]
  y=coordsn$Y[xx]
  z=values #values[xx]
  data.interp <- interp(x,y,z, duplicate="mean"
                        ,xo=seq(min(x), max(x), length = xlength)
                        ,yo=seq(min(y), max(y), length = ylength)
                        , extrap=FALSE
                        , linear=TRUE
  )
  return(data.interp)
}
